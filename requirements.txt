# Current, mutually-tested library cohort for Python 3.13 (stock Google Colab).
#
# This replaces the original late-2024 cohort (transformers 4.46 / numpy<2 / Python
# 3.10). That stack could not run on current Colab: Colab now ships Python 3.13, and
# numpy<2 / tokenizers<0.23 have no 3.13 wheels, while the condacolab route to force 3.10
# no longer works. These pins were resolved together by pip and all declare Python >=3.10
# support, so they install natively on Colab's 3.13 with no conda gymnastics.
#
# See the spec's compatibility policy (section 1) and migration guide (section 11).

transformers==5.19.0
datasets==4.8.5
tokenizers==0.23.2
# transformers 5.19 requires huggingface_hub>=1.0 — do not pin below 1.0.
huggingface_hub==1.33.0
accelerate==1.15.0
evaluate==0.4.6
# seqeval 1.2.2 ships only an sdist; its legacy setup.py build fails against very new
# setuptools_scm on Python 3.13 (ModuleNotFoundError: vcs_versioning). Install it with an
# isolated PEP 517 build:  pip install --use-pep517 seqeval==1.2.2
seqeval==1.2.2

# numpy 2.x — the modern cohort is built against the NumPy 2 ABI. Held at 2.2.6 (not the
# newest 2.5) so it also satisfies Colab's preinstalled numba (which requires numpy<2.3);
# this avoids a resolver conflict warning on Colab.
numpy==2.2.6

# torch is intentionally NOT pinned here: on Colab, use the CUDA-matched build the
# runtime ships (transformers 5.x requires torch >= 2.1). Reinstalling torch on Colab
# commonly breaks the CUDA driver match. For local (CPU) development you may
# `pip install torch` separately.
